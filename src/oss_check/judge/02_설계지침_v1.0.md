### 한 장 요약

- 구조는 PyPA 권장 `src/` 배치를 씀
- 의존은 안쪽(`domain`)으로만 향함. 외부 API 인터페이스(포트)도 `domain`에 둠
- 디자인 패턴은 네 개만 씀: 함수 목록 전략 2곳, 데코레이터, 포트·어댑터
- SOLID는 원칙마다 수위를 다르게 정함. 과설계를 막으려는 것임
- 강의 요소는 **들어 있는지**만 관리함. 적용 시점은 묻지 않음
- 이 문서의 코드 예시는 전부 실제로 실행해 테스트 11개 통과를 확인한 것임

# 4. 모듈 간 자료 구조

`domain/models.py` 가 팀의 계약임. 바꾸려면 전원 승인이 필요함.

```python
# [강의] Python Data Model / 객체 참조·가변성
#   - 모듈 사이로 넘기는 객체는 frozen=True 로 둠. 중간에 바뀌면 추적이 어려움
from dataclasses import dataclass, field
from enum import IntEnum
from typing import Literal

class Tier(IntEnum):
    PUBLIC_DOMAIN = 0
    PERMISSIVE = 1
    WEAK_COPYLEFT = 2
    STRONG_COPYLEFT = 3
    NETWORK_COPYLEFT = 4

Confidence = Literal["확실", "추정", "미상"]
Verdict = Literal["안 됨", "모름", "조건 있음", "판단보류", "써도 됨"]

# FR-OUT-02 정렬 순서. 작을수록 위험
RISK_ORDER: dict[str, int] = {"안 됨": 0, "모름": 1, "조건 있음": 2, "판단보류": 3, "써도 됨": 4}

@dataclass(frozen=True)
class Package:
    name: str
    version: str | None
    direct: bool
    depth: int

@dataclass(frozen=True)
class LicenseInfo:
    package: str
    options: tuple[frozenset[str], ...] | None   # None = 미상. OR 선택지들, 각각 AND 집합
    confidence: Confidence
    origin: str
    raw: str

@dataclass(frozen=True)
class Judgement:
    package: str
    verdict: Verdict
    reason: str
    decisive_license: str | None

@dataclass(frozen=True)
class Row:
    package: Package
    license: LicenseInfo
    judgement: Judgement

@dataclass
class Report:
    project_license: str
    rows: list[Row]
    warnings: list[str] = field(default_factory=list)
```

**v0.2 대비.** 정렬 기준을 `tier`에서 `RISK_ORDER`로 바꿈. `tier`는 「모름」「판단보류」에서 값이 없어(`None`) 정렬 시 `TypeError`가 남 (실행 확인). 또 `tier`의 크기 순서는 SRS가 요구하는 위험 순서(FR-OUT-02)와 다름.

## 5.1 P1 — 해석 전략 (R2)

```python
# [강의] Design Pattern with First-Class Functions
#   - 전략 패턴을 클래스 계층 대신 "함수 목록"으로 구현함 (교재 『전문가를 위한 파이썬』 10장)
#   - 새 단계를 추가할 때 resolve() 본문을 고치지 않음 — 개방-폐쇄 원칙 적용 지점
from collections.abc import Callable
from oss_check.domain.models import LicenseInfo
from oss_check.data_files import load_json

ALIASES: dict[str, str] = load_json("aliases.json")
Strategy = Callable[[str, dict], LicenseInfo | None]

def _canon(text: str) -> str | None:
    t = " ".join(text.split("\n")[0].split()).strip()
    return ALIASES.get(t.lower(), t if t in ("MIT", "Apache-2.0", "BSD-3-Clause") else None)

def from_expression(name: str, info: dict) -> LicenseInfo | None:
    raw = info.get("license_expression")
    if not raw:
        return None
    return LicenseInfo(name, (frozenset({raw}),), "확실", "expression", raw)

def from_license_field(name: str, info: dict) -> LicenseInfo | None:
    raw = (info.get("license") or "").strip()
    if not raw or len(raw) > 200:          # FR-LIC-02
        return None
    spdx = _canon(raw)
    return LicenseInfo(name, (frozenset({spdx}),), "확실", "license", raw) if spdx else None

def from_classifier(name: str, info: dict) -> LicenseInfo | None:
    for c in info.get("classifiers", []):
        if c.startswith("License ::") and (spdx := _canon(c.split("::")[-1].strip())):
            return LicenseInfo(name, (frozenset({spdx}),), "추정", "classifier", c)   # FR-LIC-08
    return None

# [강의] First-Class Functions — 함수를 값으로 목록에 담음
STRATEGIES: list[Strategy] = [from_expression, from_license_field, from_classifier]

def resolve(name: str, info: dict) -> LicenseInfo:
    for strategy in STRATEGIES:            # FR-LIC-01: 처음 변환에 성공한 값
        if (result := strategy(name, info)) is not None:
            return result
    return LicenseInfo(name, None, "미상", "none", "")
```

위 `_canon()`은 골격용 축약임. 실제 구현은 SRS FR-LIC-03의 정규화 전체와 별칭 사전 80종 이상, 표현식 파서(`resolve/expression.py`)를 포함해야 함.

## 5.2 P2 — 판정 규칙 (R3)

```python
# [강의] 전략 패턴 — 규칙 하나 = callable 하나
# [강의] 사용자 정의 Callable — TierRule 은 등급표를 들고 있어야 해서 __call__ 객체로 둠
# 규칙 순서가 판정을 바꿈 (SRS 3.2.4). 약한 공개조건 규칙이 등급 비교보다 앞서야 함
from collections.abc import Callable
from oss_check.domain.models import Judgement, LicenseInfo
from oss_check.data_files import load_json

TIERS: dict[str, int] = load_json("tiers.json")
Rule = Callable[[LicenseInfo, int], Judgement | None]

def rule_unknown(lic: LicenseInfo, project_tier: int) -> Judgement | None:          # FR-JDG-07
    if lic.options is None:
        return Judgement(lic.package, "모름", "표기가 없는 코드는 사용 허가가 없는 상태임", None)
    return None

class TierRule:
    def __init__(self, tiers: dict[str, int]):
        self._tiers = tiers

    def __call__(self, lic: LicenseInfo, project_tier: int) -> Judgement | None:
        assert lic.options is not None
        best: tuple[int, str] | None = None
        for combo in lic.options:                         # OR = 가장 약한 선택지 (FR-JDG-03)
            tiers = [self._tiers.get(x) for x in combo]
            if None in tiers:
                continue
            top = max(zip(tiers, sorted(combo)))         # AND = 가장 센 것 (FR-JDG-04)
            if best is None or top[0] < best[0]:
                best = top
        if best is None:
            return Judgement(lic.package, "판단보류", "등급표에 없는 라이선스임", None)  # FR-JDG-08
        tier, spdx = best
        if tier == 2 and project_tier <= 1:              # FR-JDG-05 가 FR-JDG-06 보다 먼저
            return Judgement(lic.package, "조건 있음", f"{spdx}: 고쳐 썼다면 고친 부분만 공개", spdx)
        if tier > project_tier:                          # FR-JDG-06
            return Judgement(lic.package, "안 됨", f"{spdx} 조건이 내 프로젝트보다 셈", spdx)
        return Judgement(lic.package, "써도 됨", "", spdx)

RULES: list[Rule] = [rule_unknown, TierRule(TIERS)]

# [강의] 키워드 전용 인수 — project 는 반드시 이름으로 넘김. lic 는 위치 인수로 둬서 map 에 바로 씀
def judge(lic: LicenseInfo, *, project: str) -> Judgement:
    project_tier = TIERS.get(project, 1)
    for rule in RULES:
        if (result := rule(lic, project_tier)) is not None:
            return result
    return Judgement(lic.package, "써도 됨", "", None)
```

**v0.2 대비 두 가지를 고침.**

| 문제 | v0.2 | v1.0 |
|---|---|---|
| 같은 역할에 구현이 둘 | §5.3은 클로저 `make_tier_rule`, §7.4는 `TierRule` 클래스 | `TierRule` 하나로 통일. 클로저는 P3에서 다룸 |
| `partial` + `map` 실행 오류 | `judge(*, project, license)` 에 `partial(judge, project=...)` 후 `map` → `TypeError` (실행 확인) | `judge(lic, *, project)` — 판정 대상은 위치 인수, 프로젝트는 키워드 전용 |

매개변수 이름도 `license` → `lic`로 바꿈. `license`는 파이썬 내장 이름이라 가려짐.

## 5.3 P3 — 캐시·재시도 데코레이터 (R4)

```python
# [강의] Function Decorators and Closures
#   - store 딕셔너리를 클로저가 가둬 캐시를 유지함
#   - 캐시·재시도를 호출부에 섞지 않으려고 데코레이터로 분리함
import functools
import time

def cached(ttl_sec: int = 86_400):
    def decorator(fn):
        store: dict = {}
        @functools.wraps(fn)
        def wrapper(*args):
            now = time.time()
            if args in store and now - store[args][0] < ttl_sec:
                return store[args][1]
            value = fn(*args)
            store[args] = (now, value)
            return value
        return wrapper
    return decorator

def retry(times: int = 2, backoff: float = 0.5, exceptions: tuple = (Exception,)):
    def decorator(fn):
        @functools.wraps(fn)
        def wrapper(*args, **kwargs):
            for attempt in range(times + 1):
                try:
                    return fn(*args, **kwargs)
                except exceptions:
                    if attempt == times:
                        raise
                    time.sleep(backoff * 2 ** attempt)
        return wrapper
    return decorator
```

실제 어댑터에는 이렇게 붙임.

```python
@cached(ttl_sec=86_400)                       # IF-SW-12
@retry(times=2, exceptions=(requests.Timeout, requests.ConnectionError))   # IF-SW-11
def _get_json(url: str) -> dict | None: ...
```

## 5.4 P4 — 포트·어댑터 (R4)

```python
# [강의] Interface — From Protocols to ABCs / 클린 아키텍처
#   - 포트(추상 인터페이스)는 그것을 쓰는 안쪽 계층에 둠. 바깥 어댑터가 이것을 구현함
#   - 표준 라이브러리 abc 만 import 하므로 domain 의 '외부 의존 0' 규칙을 지킴
from abc import ABC, abstractmethod

class PackageSource(ABC):
    """패키지 메타데이터 출처. 테스트에서는 가짜로 갈아끼움"""
    @abstractmethod
    def fetch(self, name: str) -> dict | None: ...

class VulnSource(ABC):
    """취약점 출처"""
    @abstractmethod
    def query(self, packages: list[tuple[str, str]]) -> dict[str, list[dict]]: ...
```

```python
# [강의] 테스트 패턴 — 외부 API 없이 테스트하기 위한 가짜 어댑터
from oss_check.domain.ports import PackageSource

class FakePackageSource(PackageSource):
    def __init__(self, data: dict[str, dict]):
        self._data = data
        self.calls = 0
    def fetch(self, name: str) -> dict | None:
        self.calls += 1
        return self._data.get(name)
```

**ABC는 이 두 개만 만듦.** 세 번째가 필요해 보이면 팀장과 먼저 상의함.

## 5.5 조정자 (R0)

```python
# [강의] 애플리케이션 계층 — 모듈 호출 순서만 조정함. 판정 로직은 여기 없음
from functools import partial
from operator import attrgetter
from oss_check.collect.walker import walk
from oss_check.domain.models import RISK_ORDER, Report, Row
from oss_check.domain.ports import PackageSource
from oss_check.judge.rules import judge
from oss_check.resolve.strategies import resolve

def analyze(roots: list[str], *, project: str, source: PackageSource) -> Report:
    packages = walk(roots, source)
    licenses = [resolve(p.name, source.fetch(p.name) or {}) for p in packages]

    # [강의] functools.partial — 프로젝트 라이선스를 고정해 1인수 callable 로 만듦
    judge_for_project = partial(judge, project=project)
    # [강의] Higher-Order Functions — map 에 함수를 넘김
    judgements = list(map(judge_for_project, licenses))

    rows = [Row(p, l, j) for p, l, j in zip(packages, licenses, judgements)]
    # [강의] operator.attrgetter — 이름순으로 먼저 정렬
    rows.sort(key=attrgetter("package.name"))
    # [강의] lambda — 위험 순으로 다시 정렬. sort 는 안정 정렬이라 같은 위험도 안에서는 이름순 유지
    rows.sort(key=lambda r: RISK_ORDER[r.judgement.verdict])   # FR-OUT-02
    return Report(project, rows)
```

## 5.6 의존성 전개 (R1)

```python
# 의존성 전개. PackageSource 를 주입받음 — 의존성 역전(D) 적용 지점
import re
from packaging.requirements import Requirement
from oss_check.domain.models import Package
from oss_check.domain.ports import PackageSource

ENV = {"python_version": "3.11", "sys_platform": "linux", "platform_system": "Linux",
       "os_name": "posix", "platform_machine": "x86_64", "implementation_name": "cpython",
       "python_full_version": "3.11.9", "platform_python_implementation": "CPython"}

def normalize(name: str) -> str:                     # FR-DEP-04 (PEP 503)
    return re.sub(r"[-_.]+", "-", name).lower()

def walk(roots: list[str], source: PackageSource) -> list[Package]:
    seen: dict[str, Package] = {}
    queue = [(r, 0, frozenset()) for r in roots]
    while queue:
        raw, depth, parent_extras = queue.pop(0)
        req = Requirement(raw)
        if req.marker:
            envs = [dict(ENV, extra=e) for e in parent_extras] or [dict(ENV, extra="")]
            if not any(req.marker.evaluate(env) for env in envs):   # FR-DEP-02, FR-DEP-03
                continue
        name = normalize(req.name)
        if name in seen:                                # FR-DEP-05, FR-DEP-06
            continue
        seen[name] = Package(name, None, depth == 0, depth)
        info = source.fetch(name)
        for dep in (info or {}).get("requires_dist") or []:
            queue.append((dep, depth + 1, frozenset(req.extras)))
    return list(seen.values())
```

**구현 시 주의.** 같은 패키지가 먼저 extras 없이 들어오고 나중에 extras와 함께 다시 나오면, 위 골격은 두 번째를 건너뜀. 실제 구현에서는 패키지별로 「요청된 extras 집합」을 누적해 새 extras가 생기면 다시 전개해야 함 (FR-DEP-03).

## 5.7 쓰지 않을 패턴

| 패턴 | 이유 |
|---|---|
| Factory | 만들 객체 종류가 적음 |
| Observer | 이벤트가 없음 |
| Singleton | 파이썬 모듈이 이미 그 역할을 함 |
| Repository | DB가 없음 |
| 세 번째 이상의 ABC | 2학년 6인 팀이 관리하기 어려움 |

# 6. SOLID 적용 수위

수위를 나누는 이유는 과설계 방지임. 정한 수위는 첫 커밋부터 지킴.

| 원칙 | 수위 | 적용 범위 | 확인 |
|---|---|---|---|
| S 단일 책임 | 전면 | 모든 모듈·함수 | 함수 50줄, 파일 300줄 초과 시 분리 |
| O 개방-폐쇄 | 부분 | `STRATEGIES`, `RULES` 두 곳 | 테스트 `test_new_strategy_needs_no_change_to_resolve` |
| L 리스코프 치환 | 해당 없음 | — | 상속 깊이 2단 초과 금지 |
| I 인터페이스 분리 | 최소 | ABC 두 개, 각 메서드 하나 | ABC 개수 2개 초과 금지 |
| D 의존성 역전 | 부분 | 외부 API만 (포트·어댑터) | `domain` import 검사 테스트 |

### 과설계 신호

| 신호 | 대응 |
|---|---|
| 구현체가 하나뿐인 인터페이스 | 지움 (단, 포트 2개는 가짜 구현이 있으므로 예외) |
| `Base`·`Abstract`·`Manager`로 끝나는 이름이 늘어남 | 멈추고 합침 |
| 기능 하나에 파일 5개 이상 수정 | 경계를 다시 봄 |
| 「나중에 확장될지 몰라서」 | 지금 필요 없으면 안 만듦 |

### PR 템플릿에 넣을 항목

```
- [ ] 새 ABC를 만들지 않았음 (만들었다면 이유)
- [ ] domain/ 이 표준 라이브러리 외에는 import 하지 않음
- [ ] 함수 50줄, 파일 300줄 이하
```

### SOLID 수업 이후

이미 적용해 둔 것을 수업의 정의와 대조하고, 결과를 `docs/solid-review.md` 한 장으로 남김. 리팩터링이 아니라 대조임.
